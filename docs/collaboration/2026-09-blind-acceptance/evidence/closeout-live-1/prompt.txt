+@app.route("/api/projects/<project_id>/clips/import_preview", methods=["POST"])
+def import_preview(project_id):
+    projects = _load_projects()
+    project = projects.get(project_id)
+    if not project:
+        return jsonify({"error": "Project not found"}), 404
+    if "file" not in request.files:
+        return jsonify({"error": "No file provided", "code": "bad_request"}), 400
+    try:
+        text = request.files["file"].read().decode("utf-8-sig")
+    except UnicodeDecodeError:
+        return jsonify({"error": "Undecodable file", "code": "bad_request"}), 400
+    reader = csv.reader(io.StringIO(text))
+    header = next(reader, [])
+    col = {h.strip().lower(): i for i, h in enumerate(header)}
+    if not {"tag type", "start (s)", "end (s)"} <= col.keys():
+        return jsonify({"error": "Missing required header", "code": "bad_request"}), 400
+    n, known = len(header), {t["name"] for t in project["tag_types"]}
+    rows, total, n_valid, n_mal = [], 0, 0, 0
+
+    def cell(rec, name):
+        i = col.get(name)
+        return rec[i].strip() if i is not None and i < len(rec) else ""
+
+    def parse_num(rec, name):
+        try:
+            value = float(cell(rec, name))
+            return value if math.isfinite(value) else None
+        except ValueError:
+            return None
+
+    while True:
+        line = reader.line_num + 1
+        rec = next(reader, None)
